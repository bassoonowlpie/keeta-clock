<script>
const KEY = 'keeta_walker_records';
const CFG = 'keeta_walker_cfg';
const DELETED_KEY = 'keeta_walker_deleted';
const BONUS_KEY = 'keeta_walker_bonuses';
let records = JSON.parse(localStorage.getItem(KEY) || '[]');
let openRec = JSON.parse(localStorage.getItem(KEY + '_open') || 'null');
let deletedIds = JSON.parse(localStorage.getItem(DELETED_KEY) || '[]');
let bonuses = JSON.parse(localStorage.getItem(BONUS_KEY) || '[]');
let cfg = null, cloudSha = null;
const todayStr = () => { const d = new Date(); return d.getFullYear() + '-' + String(d.getMonth()+1).padStart(2,'0') + '-' + String(d.getDate()).padStart(2,'0'); };
const wday = (d) => '日一二三四五六'[new Date(d).getDay()];
const fmtTime = (t) => { const d = new Date(t); return String(d.getHours()).padStart(2,'0') + ':' + String(d.getMinutes()).padStart(2,'0'); };
const fmtDur = (ms) => { const m = Math.floor(ms/60000); const h = Math.floor(m/60); return h + '小時' + (m%60) + '分'; };
const uid = () => Date.now().toString(36) + Math.random().toString(36).slice(2);
function saveLocal() { localStorage.setItem(KEY, JSON.stringify(records)); localStorage.setItem(KEY + '_open', JSON.stringify(openRec)); localStorage.setItem(DELETED_KEY, JSON.stringify(deletedIds)); localStorage.setItem(BONUS_KEY, JSON.stringify(bonuses)); }
function msg(text, type) { const el = document.getElementById('msg'); el.textContent = text; el.className = type || ''; if (text) setTimeout(() => { el.textContent = ''; el.className = ''; }, 3000); }
function getLastRecordIdPerDay() { const map = {}; records.forEach(r => { map[r.date] = r.id; }); return map; }
function dailyIncomeMap() { const map = {}; records.forEach(r => { if (!map[r.date]) map[r.date] = 0; map[r.date] = Math.max(map[r.date], r.money || 0); }); return map; }
function getDayBonus(date) { return bonuses.filter(b => b.date === date).reduce((s, b) => s + b.amount, 0); }
function getLastMoney() { const last = records.filter(r => r.money != null).pop(); return last ? last.money : null; }
function updateLastMoneyHint() { const hint = document.getElementById('lastMoneyHint'); const last = getLastMoney(); hint.textContent = last !== null ? '上次收工金額：HK$' + last.toFixed(1) + '（可作參考）' : ''; }
function makeCalLink(startMs, endMs, money) { const s = new Date(startMs); const e = new Date(endMs); const fmt = (d) => d.toISOString().replace(/[-:]/g,'').split('.')[0] + 'Z'; return 'https://www.google.com/calendar/render?action=TEMPLATE&text=Keeta+Walker+收工&dates=' + fmt(s) + '/' + fmt(e) + '&details=今日收入+HK$' + (money||0).toFixed(1); }
function parseLocalDate(d) { return new Date(d + 'T00:00:00'); }
async function apiGet(path) { const res = await fetch('https://api.github.com/repos/' + cfg.user + '/' + cfg.repo + '/contents/' + path, { headers: { Authorization: 'token ' + cfg.token, Accept: 'application/vnd.github.v3+json' } }); if (res.status === 404) return null; if (!res.ok) throw new Error('HTTP ' + res.status); return res.json(); }
async function apiPut(path, body) { const res = await fetch('https://api.github.com/repos/' + cfg.user + '/' + cfg.repo + '/contents/' + path, { method: 'PUT', headers: { Authorization: 'token ' + cfg.token, Accept: 'application/vnd.github.v3+json', 'Content-Type': 'application/json' }, body: JSON.stringify(body) }); if (!res.ok) throw new Error('HTTP ' + res.status); return res.json(); }
async function cloudLoad() { try { const data = await apiGet('data.json'); if (!data) { msg('雲端沒有資料，開始使用本機資料'); return; } const json = JSON.parse(atob(data.content.replace(/\s/g,''))); if (json.deleted && Array.isArray(json.deleted)) { json.deleted.forEach(id => { if (!deletedIds.includes(id)) deletedIds.push(id); }); } if (json.bonuses && Array.isArray(json.bonuses)) { bonuses = json.bonuses; localStorage.setItem(BONUS_KEY, JSON.stringify(bonuses)); } const cloudRecs = (json.records || []).filter(r => !deletedIds.includes(r.id)); const localIds = new Set(records.map(r => r.id)); const merged = [...records]; cloudRecs.forEach(r => { if (!localIds.has(r.id)) merged.push(r); }); merged.sort((a, b) => a.in - b.in); records = merged; saveLocal(); refresh(); msg('☁️ 已從雲端同步', 'ok'); } catch (e) { msg('雲端載入失敗：' + e.message, 'err'); } }
async function cloudSave() { try { const payload = { records, deleted: deletedIds, bonuses }; const content = btoa(unescape(encodeURIComponent(JSON.stringify(payload)))); const body = { message: 'update data.json', content, sha: cloudSha }; const res = await apiPut('data.json', body); cloudSha = res.content.sha; msg('☁️ 已同步到雲端', 'ok'); document.getElementById('lastSync').textContent = new Date().toLocaleString(); return true; } catch (e) { msg('雲端同步失敗：' + e.message, 'err'); return false; } }
async function forceSync() { if (!cfg) { alert('請先在下面設定 GitHub Token'); return; } await cloudSave(); }
async function syncFromCloud() { if (!cfg) { alert('請先在下面設定 GitHub Token'); return; } await cloudLoad(); }
function saveConfig() {
  const user = document.getElementById('cfgUser').value.trim();
  const repo = document.getElementById('cfgRepo').value.trim();
  const token = document.getElementById('cfgToken').value.trim();
  if (!user || !repo || !token) { alert('三項都要填！'); return; }
  cfg = { user, repo, token };
  localStorage.setItem(CFG, JSON.stringify(cfg));
  msg(''); cloudLoad().then(() => cloudSave());
}
function clearConfig() {
  cfg = null; cloudSha = null; deletedIds = []; bonuses = [];
  localStorage.removeItem(CFG); localStorage.removeItem(DELETED_KEY); localStorage.removeItem(BONUS_KEY);
  document.getElementById('cfgUser').value = '';
  document.getElementById('cfgRepo').value = '';
  document.getElementById('cfgToken').value = '';
  msg('已清除雲端設定，只用本機模式');
  document.getElementById('lastSync').textContent = '';
}

/* ---------- 打卡 ---------- */
async function clockIn() {
  if (openRec) { alert('已經開工了！'); return; }
  openRec = { id: uid(), date: todayStr(), in: Date.now(), out: null, money: null, note: '' };
  updateLastMoneyHint();
  saveLocal(); refresh();
  if (cfg) {
    const ok = await cloudSave();
    if (ok) alert('✅ 開工打卡成功！已同步到雲端\n' + fmtTime(openRec.in));
    else alert('✅ 開工打卡成功！（雲端同步失敗，已保留在本機）\n' + fmtTime(openRec.in));
  } else alert('✅ 開工打卡成功！' + fmtTime(openRec.in));
}

async function clockOut() {
  if (!openRec) { alert('還未開工！'); return; }
  const v = parseFloat(document.getElementById('moneyInput').value);
  if (isNaN(v) || v < 0) { alert('請輸入今日累計賺了多少錢！'); return; }
  openRec.out = Date.now();
  openRec.money = Math.round(v * 10) / 10;
  openRec.note = document.getElementById('noteInput').value.trim();
  records.push(openRec);
  openRec = null;
  document.getElementById('moneyInput').value = '';
  document.getElementById('noteInput').value = '';
  saveLocal(); refresh();
  if (cfg) {
    const ok = await cloudSave();
    if (ok) alert('✅ 收工打卡成功！已同步到雲端\n今日累計賺了 HK$' + v);
    else alert('✅ 收工打卡成功！（雲端同步失敗，已保留在本機）\n今日累計賺了 HK$' + v);
  } else alert('✅ 收工打卡成功！今日累計賺了 HK$' + v);
}

async function delRec(i) {
  if (!confirm('確定刪除這條記錄？')) return;
  const id = records[i].id;
  records.splice(i, 1);
  if (!deletedIds.includes(id)) deletedIds.push(id);
  saveLocal(); refresh();
  if (cfg) { const ok = await cloudSave(); if (ok) msg('🗑 已刪除並同步到雲端', 'ok'); else msg('🗑 已刪除（雲端同步失敗）', 'err'); }
  else msg('🗑 已刪除', 'ok');
}

async function saveEdit(i) {
  const r = records[i];
  const inVal = document.getElementById('editIn').value;
  const outVal = document.getElementById('editOut').value;
  const moneyVal = document.getElementById('editMoney').value;
  const noteVal = document.getElementById('editNote').value;
  
  if (!inVal) { alert('開工時間不能為空！'); return; }
  
  const [inH, inM] = inVal.split(':').map(Number);
  const inDate = new Date(r.in);
  inDate.setHours(inH, inM, 0, 0);
  r.in = inDate.getTime();
  
  if (outVal) {
    const [outH, outM] = outVal.split(':').map(Number);
    const outDate = new Date(r.out || r.in);
    outDate.setHours(outH, outM, 0, 0);
    r.out = outDate.getTime();
  } else {
    r.out = null;
  }
  
  if (moneyVal !== '') {
    r.money = Math.round(parseFloat(moneyVal) * 10) / 10;
  } else {
    r.money = null;
  }
  
  r.note = noteVal.trim();
  
  closeEdit();
  saveLocal(); refresh();
  if (cfg) { const ok = await cloudSave(); if (ok) msg('✅ 已儲存並同步到雲端', 'ok'); else msg('✅ 已儲存（雲端同步失敗）', 'err'); }
  else msg('✅ 已儲存', 'ok');
}

/* ---------- 統計 ---------- */
function calcStats(rangeDays) {
  const end = new Date(); const start = new Date(); start.setDate(start.getDate() - rangeDays + 1);
  const days = new Set(); let totalMin = 0; let totalMoney = 0;
  records.forEach(r => {
    const d = new Date(r.in);
    if (d >= start && d <= end) {
      days.add(r.date);
      if (r.out) totalMin += (r.out - r.in) / 60000;
    }
  });
  const incMap = dailyIncomeMap();
  Object.keys(incMap).forEach(d => {
    const date = new Date(d + 'T00:00:00');
    if (date >= start && date <= end) totalMoney += incMap[d];
  });
  bonuses.forEach(b => {
    const date = new Date(b.date + 'T00:00:00');
    if (date >= start && date <= end) totalMoney += b.amount;
  });
  return { days: days.size, sessions: records.filter(r => { const d = new Date(r.in); return d >= start && d <= end; }).length, hours: (totalMin/60).toFixed(1), money: totalMoney.toFixed(1) };
}
function updateStats() { const s7 = calcStats(7); const s30 = calcStats(30); const el = document.getElementById('stats'); if (!s7.sessions && !s30.sessions) { el.innerHTML = ''; return; } el.innerHTML = '<div style="background:#f0f2f5;border-radius:12px;padding:12px 16px;margin-bottom:12px;font-size:14px;"><strong>📊 統計</strong> ' + '近7日：' + s7.days + '天 / ' + s7.sessions + '段 / ' + s7.hours + '小時 / HK$' + s7.money + '　' + '近30日：' + s30.days + '天 / ' + s30.sessions + '段 / ' + s30.hours + '小時 / HK$' + s30.money + '</div>'; }

function calInit() {
  const now = new Date(); let y = now.getFullYear(); let m = now.getMonth();
  const box = document.getElementById('calBox');
  const title = document.getElementById('calTitle');
  function render() {
    title.textContent = y + '年' + (m+1) + '月';
    const first = new Date(y, m, 1); const last = new Date(y, m+1, 0);
    const startDay = first.getDay(); const days = last.getDate();
    const incMap = dailyIncomeMap();
    let html = '<table class="cal-table"><tr><th>日</th><th>一</th><th>二</th><th>三</th><th>四</th><th>五</th><th>六</th></tr><tr>';
    for (let i = 0; i < startDay; i++) html += '<td></td>';
    for (let d = 1; d <= days; d++) {
      const date = y + '-' + String(m+1).padStart(2,'0') + '-' + String(d).padStart(2,'0');
      const has = records.some(r => r.date === date);
      const money = incMap[date] || 0;
      const bonus = getDayBonus(date);
      const cls = has ? 'has' : '';
      const amt = money + bonus;
      html += '<td class="' + cls + '" onclick="calClick(\'' + date + '\')">' + d + (amt > 0 ? '<div class="cal-amt">$' + amt.toFixed(0) + '</div>' : '') + '</td>';
      if ((startDay + d) % 7 === 0) html += '</tr><tr>';
    }
    html += '</tr></table>';
    box.innerHTML = html;
  }
  render();
  document.getElementById('calPrev').onclick = () => { m--; if (m < 0) { m = 11; y--; } render(); };
  document.getElementById('calNext').onclick = () => { m++; if (m > 11) { m = 0; y++; } render(); };
}

/* ---------- 搜尋 ---------- */
function doSearch() {
  const q = document.getElementById('searchInput').value.trim();
  if (!q) { view = 'day'; renderRecords(); return; }
  view = 'search';
  const lower = q.toLowerCase();
  const matched = records.filter(r => {
    return r.date.includes(q) || (r.note && r.note.toLowerCase().includes(lower)) || fmtTime(r.in).includes(q) || (r.out && fmtTime(r.out).includes(q));
  });
  const matchedBonuses = bonuses.filter(b => b.date.includes(q) || (b.label && b.label.toLowerCase().includes(lower)));
  renderSearchResults(matched, matchedBonuses, q);
}
function renderSearchResults(matched, matchedBonuses, q) {
  const box = document.getElementById('recordBox');
  if (!matched.length && !matchedBonuses.length) { box.innerHTML = '<div class="empty">沒有找到符合「' + q + '」的記錄</div>'; return; }
  let html = '<div style="margin-bottom:8px;font-size:14px;color:#555;">🔍 找到 ' + matched.length + ' 條記錄' + (matchedBonuses.length ? '、' + matchedBonuses.length + ' 筆獎賞' : '') + '</div>';
  const dayMap = {};
  matched.forEach(r => { if (!dayMap[r.date]) dayMap[r.date] = []; dayMap[r.date].push(r); });
  Object.keys(dayMap).sort().reverse().forEach(date => {
    const recs = dayMap[date]; const idxs = recs.map(r => records.indexOf(r));
    html += buildDayCard(date, recs, idxs, true);
  });
  matchedBonuses.forEach(b => {
    html += '<div class="bonus-card"><span class="gift">🎁</span><div class="bc-info"><div class="bc-date">' + b.date + '</div><div class="bc-label">' + b.label + '</div></div><span class="bc-amt">$' + b.amount.toFixed(1) + '</span></div>';
  });
  box.innerHTML = html;
}

function buildDayCard(date, dayRecs, dayIdxs, isSearchResult) {
  const lastRec = dayRecs[dayRecs.length - 1];
  const dayBonusAmt = getDayBonus(date);
  const today = todayStr();
  const isToday = date === today;

  let sessionsHtml = '';
  dayRecs.forEach((r, i) => {
    const idx = dayIdxs[i];
    const dur = r.out ? fmtDur(r.out - r.in) : '-';
    const calLink = r.out ? '<a class="session-cal" href="' + makeCalLink(r.in, r.out, r.money) + '" target="_blank" title="加入行事曆">📅</a>' : '';
    const delBtn = '<button class="session-del" onclick="delRec(' + idx + ')" title="刪除">✕</button>';
    const editBtn = '<button class="session-edit" onclick="editRec(' + idx + ')" title="編輯">✎</button>';
    const noteBadge = r.note ? '<span class="session-note">' + r.note + '</span>' : '';
    sessionsHtml += '<div class="session-row">' +
      '<div class="session-left">' +
        '<div class="session-time">' +
          '<span class="dot in"></span>' +
          '<span class="time-str">' + fmtTime(r.in) + '</span>' +
          '<span class="arrow">→</span>' +
          '<span class="time-str">' + (r.out ? fmtTime(r.out) : '--:--') + '</span>' +
          '<span class="dur">(' + dur + ')</span>' +
        '</div>' +
        noteBadge +
      '</div>' +
      '<div class="session-actions">' +
        (r.money != null ? '<span class="session-money">$' + r.money.toFixed(1) + '</span>' : '') +
        calLink +
        editBtn +
        delBtn +
      '</div>' +
    '</div>';
  });

  let bonusHtml = '';
  const dayBonuses = bonuses.filter(b => b.date === date);

function renderRecords() {
  const box = document.getElementById('recordBox');
  if (!records.length && !bonuses.length) { box.innerHTML = '<div class="empty">還沒有記錄，快開工打卡吧！</div>'; return; }

  if (view === 'day') {
    // Group records by date
    const dayMap = {};
    records.forEach((r, idx) => {
      if (!dayMap[r.date]) dayMap[r.date] = { recs: [], idxs: [] };
      dayMap[r.date].recs.push(r);
      dayMap[r.date].idxs.push(idx);
    });
    let html = '';
    Object.keys(dayMap).sort().reverse().forEach(date => {
      html += buildDayCard(date, dayMap[date].recs, dayMap[date].idxs, false);
    });
    box.innerHTML = html;
  }
}
