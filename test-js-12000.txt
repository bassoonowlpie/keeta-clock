async function saveEdit(i) {
  const r = records[i];
  const inVal = document.getElementById('editIn').value;
  const outVal = document.getElementById('editOut').value;
  const moneyVal = document.getElementById('editMoney').value;
  const noteVal = document.getElementById('editNote').value;
  
  if (!inVal) { alert('開工時間不能為空！'); return; }
  
  const [inH, inM] = inVal.split(':').map(Number);
  const inDate = new Date(r.in);
  inDate.setHours(inH, inM, 0, 0);
  r.in = inDate.getTime();
  
  if (outVal) {
    const [outH, outM] = outVal.split(':').map(Number);
    const outDate = new Date(r.out || r.in);
    outDate.setHours(outH, outM, 0, 0);
    r.out = outDate.getTime();
  } else {
    r.out = null;
  }
  
  if (moneyVal !== '') {
    r.money = Math.round(parseFloat(moneyVal) * 10) / 10;
  } else {
    r.money = null;
  }
  
  r.note = noteVal.trim();
  
  closeEdit();
  saveLocal(); refresh();
  if (cfg) { const ok = await cloudSave(); if (ok) msg('✅ 已儲存並同步到雲端', 'ok'); else msg('✅ 已儲存（雲端同步失敗）', 'err'); }
  else msg('✅ 已儲存', 'ok');
}

function exportCSV() {

    csv += b.date + ',,,,,' + '獎賞' + ',' + b.label + '\r\n';
  });

}

/* ---------- 統計 ---------- */
function calcStats(rangeDays) {
  const end = new Date(); const start = new Date(); start.setDate(start.getDate() - rangeDays + 1);
  const days = new Set(); let totalMin = 0; let totalMoney = 0;
  records.forEach(r => {
    const d = new Date(r.in);
    if (d >= start && d <= end) {
      days.add(r.date);
      if (r.out) totalMin += (r.out - r.in) / 60000;
    }
  });
  const incMap = dailyIncomeMap();
  Object.keys(incMap).forEach(d => {
    const date = new Date(d + 'T00:00:00');
    if (date >= start && date <= end) totalMoney += incMap[d];
  });
  bonuses.forEach(b => {
    const date = new Date(b.date + 'T00:00:00');
    if (date >= start && date <= end) totalMoney += b.amount;
  });
  return { days: days.size, sessions: records.filter(r => { const d = new Date(r.in); return d >= start && d <= end; }).length, hours: (totalMin/60).toFixed(1), money: totalMoney.toFixed(1) };
}
function updateStats() { const s7 = calcStats(7); const s30 = calcStats(30); const el = document.getElementById('stats'); if (!s7.sessions && !s30.sessions) { el.innerHTML = ''; return; } el.innerHTML = '<div style="background:#f0f2f5;border-radius:12px;padding:12px 16px;margin-bottom:12px;font-size:14px;"><strong>📊 統計</strong> ' + '近7日：' + s7.days + '天 / ' + s7.sessions + '段 / ' + s7.hours + '小時 / HK$' + s7.money + '　' + '近30日：' + s30.days + '天 / ' + s30.sessions + '段 / ' + s30.hours + '小時 / HK$' + s30.money + '</div>'; }

function calInit() {
  const now = new Date(); let y = now.getFullYear(); let m = now.getMonth();
  const box = document.getElementById('calBox');
  const title = document.getElementById('calTitle');
  function render() {
    title.textContent = y + '年' + (m+1) + '月';
    const first = new Date(y, m, 1); const last = new Date(y, m+1, 0);
    const startDay = first.getDay(); const days = last.getDate();
    const incMap = dailyIncomeMap();
    let html = '<table class="cal-table"><tr><th>日</th><th>一</th><th>二</th><th>三</th><th>四</th><th>五</th><th>六</th></tr><tr>';
    for (let i = 0; i < startDay; i++) html += '<td></td>';
    for (let d = 1; d <= days; d++) {
      const date = y + '-' + String(m+1).padStart(2,'0') + '-' + String(d).padStart(2,'0');
      const has = records.some(r => r.date === date);
      const money = incMap[date] || 0;
      const bonus = getDayBonus(date);
      const cls = has ? 'has' : '';
      const amt = money + bonus;
      html += '<td class="' + cls + '" onclick="calClick(\'' + date + '\')">' + d + (amt > 0 ? '<div class="cal-amt">$' + amt.toFixed(0) + '</div>' : '') + '</td>';
      if ((startDay + d) % 7 === 0) html += '</tr><tr>';
    }
    html += '</tr></table>';
    box.innerHTML = html;
  }
  render();
  document.getElementById('calPrev').onclick = () => { m--; if (m < 0) { m = 11; y--; } render(); };
  document.getElementById('calNext').onclick = () => { m++; if (m > 11) { m = 0; y++; } render(); };
}

function doSearch() {
  const q = document.getElementById('searchInput').value.trim();
  if (!q) { view = 'day'; renderRecords(); return; }
  view = 'search';
  const lower = q.toLowerCase();
  const matched = records.filter(r => {
    return r.date.includes(q) || (r.note && r.note.toLowerCase().includes(lower)) || fmtTime(r.in).includes(q) || (r.out && fmtTime(r.out).includes(q));
  });
  const matchedBonuses = bonuses.filter(b => b.date.includes(q) || (b.label && b.label.toLowerCase().includes(lower)));
  renderSearchResults(matched, matchedBonuses, q);
}
function renderSearchResults(matched, matchedBonuses, q) {
  const box = document.getElementById('recordBox');
  if (!matched.length && !matchedBonuses.length) { box.innerHTML = '<div class="empty">沒有找到符合「' + q + '」的記錄</div>'; return; }
  let html = '<div style="margin-bottom:8px;font-size:14px;color:#555;">🔍 找到 ' + matched.length + ' 條記錄' + (matchedBonuses.length ? '、' + matchedBonuses.length + ' 筆獎賞' : '') + '</div>';
  const dayMap = {};
  matched.forEach(r => { if (!dayMap[r.date]) dayMap[r.date] = []; dayMap[r.date].push(r); });
  Object.keys(dayMap).sort().reverse().forEach(date => {
    const recs = dayMap[date]; const idxs = recs.map(r => records.indexOf(r));
    html += buildDayCard(date, recs, idxs, true);
  });
  matchedBonuses.forEach(b => {
    html += '<div class="bonus-card"><span class="gift">🎁</span><div class="bc-info"><div class="bc-date">' + b.date + '</div><div class="bc-label">' + b.label + '</div></div><span class="bc-amt">$' + b.amount.toFixed(1) + '</span></div>';
  });
  box.innerHTML = html;
}

function buildDayCard(date, dayRecs, dayIdxs, isSearchResult) {
  const lastRec = dayRecs[dayRecs.length - 1];
  const dayBonusAmt = getDayBonus(date);
  const today = todayStr();
  const isToday = date === today;

  let sessionsHtml = '';
  dayRecs.forEach((r, i) => {
    const idx = dayIdxs[i];
    const dur = r.out ? fmtDur(r.out - r.in) : '-';
    const calLink = r.out ? '<a class="session-cal" href="' + makeCalLink(r.in, r.out, r.money) + '" target="_blank" title="加入行事曆">📅</a>' : '';
    const delBtn = '<button class="session-del" onclick="delRec(' + idx + ')" title="刪除">✕</button>';
    const editBtn = '<button class="session-edit" onclick="editRec(' + idx + ')" title="編輯">✎</button>';
    const noteBadge = r.note ? '<span class="session-note">' + r.note + '</span>' : '';
    sessionsHtml += '<div class="session-row">' +
      '<div class="session-left">' +
        '<div class="session-time">' +
          '<span class="dot in"></span>' +
          '<span class="time-str">' + fmtTime(r.in) + '</span>' +
          '<span class="arrow">→</span>' +
          '<span class="time-str">' + (r.out ? fmtTime(r.out) : '--:--') + '</span>' +
          '<span class="dur">(' + dur + ')</span>' +
        '</div>' +
        noteBadge +
      '</div>' +
      '<div class="session-actions">' +
        (r.money != null ? '<span class="session-money">$' + r.money.toFixed(1) + '</span>' : '') +
        calLink +
        editBtn +
        delBtn +
      '</div>' +
    '</div>';
  });

  let bonusHtml = '';
  const dayBonuses = bonuses.filter(b => b.date === date);
  dayBonuses.forEach(b => {
    bonusHtml += '<div class="bonus-row2">' +
      '<span class="gift">🎁</span>' +
      '<span class="bonus-label">獎賞</span>' +
      '<span class="bonus-amt">$' + b.amount.toFixed(1) + '</span>' +
    '</div>';
  });

  const totalMoney = (lastRec.money || 0) + dayBonusAmt;

  return '<div class="day-card ' + (isToday ? 'today' : '') + '">' +
    '<div class="day-header">' +

  '</div>';
}
function renderRecords() {
  const box = document.getElementById('recordBox');
  if (!records.length && !bonuses.length) { box.innerHTML = '<div class="empty">還沒有記錄，快開工打卡吧！</div>'; return; }

  if (view === 'day') {
    // Group records by date
    const dayMap = {};
    records.forEach((r, idx) => {
      if (!dayMap[r.date]) dayMap[r.date] = { recs: [], idxs: [] };
      dayMap[r.date].recs.push(r);
      dayMap[r.date].idxs.push(idx);
    });
    let html = '';
    Object.keys(dayMap).sort().reverse().forEach(date => {
      html += buildDayCard(date, dayMap[date].recs, dayMap[date].idxs, false);
    });
    box.innerHTML = html;
  }
}
